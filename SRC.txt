extends CharacterBody2D

@export var cooldown_dvojskoku: float = 1.0

@onready var animated_sprite: AnimatedSprite2D = $AnimatedSprite2D

const SPEED: float = 900.0
const JUMP_VELOCITY: float = 1100.0
const AIR_JUMP_VELOCITY: float = 950.0
const GRAVITACE: float = 1400.0
const MAX_SKOKU_VE_VZDUCHU: int = 1
const COYOTE: float = 0.12

var mleko_scena = preload("res://mleko.tscn")

var _cas_bez_zeme: float = 0.0
var _skoku_ve_vzduchu: int = 0
var _coyote: float = 0.0
var _cooldown: float = 0.0


func _ready() -> void:
	up_direction = Vector2(0, 1)
	motion_mode = MOTION_MODE_GROUNDED
	floor_snap_length = 0.0
	floor_stop_on_slope = false
	floor_constant_speed = true
	floor_block_on_wall = true
	floor_max_angle = deg_to_rad(60.0)
	slide_on_ceiling = false
	set_collision_mask_value(2, true)


func _physics_process(delta: float) -> void:
	_cooldown = maxf(0.0, _cooldown - delta)

	if is_on_floor():
		_cas_bez_zeme = 0.0
		_coyote = COYOTE
		_skoku_ve_vzduchu = 0
	else:
		_cas_bez_zeme += delta
		_coyote = maxf(0.0, _coyote - delta)
		var nasobek := lerpf(1.0, 2.2, clampf(_cas_bez_zeme / 0.3, 0.0, 1.0))
		if velocity.y > -200.0:
			nasobek *= 0.6
		velocity.y -= GRAVITACE * nasobek * delta
		velocity.y = maxf(velocity.y, -1800.0)

	if Input.is_action_just_pressed("jump"):
		if is_on_floor() or _coyote > 0.0:
			velocity.y = JUMP_VELOCITY
			_coyote = 0.0
		elif _skoku_ve_vzduchu < MAX_SKOKU_VE_VZDUCHU and _cooldown <= 0.0:
			velocity.y = AIR_JUMP_VELOCITY
			_skoku_ve_vzduchu += 1
			_cooldown = cooldown_dvojskoku
			vystrik_mleko()

	var direction := Input.get_axis("left", "right")

	if direction:
		velocity.x = direction * SPEED
		animated_sprite.play("run")
		animated_sprite.flip_h = direction < 0
		animated_sprite.scale = Vector2(1.0, 1.0)
	else:
		velocity.x = move_toward(velocity.x, 0, SPEED)
		animated_sprite.play("idle")
		animated_sprite.scale = Vector2(0.1, 0.1)
		animated_sprite.position.y = -20

	move_and_slide()


func vystrik_mleko() -> void:
	var mleko = mleko_scena.instantiate()
	get_parent().add_child(mleko)
	mleko.global_position = $MlekoPozice.global_position
	mleko.add_collision_exception_with(self)
	await get_tree().create_timer(0.3).timeout
	if is_instance_valid(mleko):
		mleko.remove_collision_exception_with(self)

extends Area2D

@export_file("*.tscn") var menu_scena: String = "res://Krava/menu.tscn"
@export var text_dokonceni: String = "LEVEL COMPLETE"
@export var doba_cekani: float = 1.4

var _hotovo: bool = false


func _ready() -> void:
	body_entered.connect(_on_body_entered)


func _on_body_entered(body: Node2D) -> void:
	if _hotovo:
		return
	if not body.is_in_group("player"):
		return
	_hotovo = true
	_ukonci_level(body)


func _ukonci_level(hrac: Node) -> void:
	set_deferred("monitoring", false)
	if hrac is CharacterBody2D:
		hrac.set_physics_process(false)

	var vrstva := CanvasLayer.new()
	vrstva.layer = 100
	add_child(vrstva)

	var kryt := ColorRect.new()
	kryt.color = Color(0.0, 0.0, 0.0, 0.0)
	kryt.mouse_filter = Control.MOUSE_FILTER_IGNORE
	vrstva.add_child(kryt)
	kryt.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)

	var stred := CenterContainer.new()
	stred.mouse_filter = Control.MOUSE_FILTER_IGNORE
	vrstva.add_child(stred)
	stred.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)

	var popis := Label.new()
	popis.text = text_dokonceni
	popis.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	popis.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	popis.mouse_filter = Control.MOUSE_FILTER_IGNORE
	popis.add_theme_font_size_override("font_size", 96)
	popis.add_theme_color_override("font_color", Color(1.0, 0.97, 0.85))
	popis.add_theme_color_override("font_outline_color", Color(0.1, 0.05, 0.0))
	popis.add_theme_constant_override("outline_size", 20)
	popis.add_theme_color_override("font_shadow_color", Color(0.0, 0.0, 0.0, 0.6))
	popis.add_theme_constant_override("shadow_offset_x", 6)
	popis.add_theme_constant_override("shadow_offset_y", 6)
	popis.modulate.a = 0.0
	stred.add_child(popis)

	var tween := create_tween()
	tween.set_parallel(true)
	tween.tween_property(popis, "modulate:a", 1.0, 0.4)
	tween.tween_property(kryt, "color:a", 1.0, 0.6).set_delay(doba_cekani - 0.5)
	tween.set_parallel(false)
	tween.tween_interval(doba_cekani)
	await tween.finished

	get_tree().change_scene_to_file(menu_scena)
